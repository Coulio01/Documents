select hash(customer_public_hash) as Hash1,customer_public_hash,valid_from
from deltaDf where customer_public_hash in 
  ( 
    select customer_public_hash from
     (  
                    select customer_public_hash,valid_from from deltaDf
                    except
                    select customer_public_hash,valid_from from custDf where customer_sk is not null
     )C
  )    
)X
'''.format(custskmaxInt=custskmaxInt)
deltadff=spark.sql(skgen_query)
deltadff.createOrReplaceTempView('deltadff')
mergeQuery = '''
merge into custDf as c
using deltadff as d
on d.`customer_public_hash` = c.`customer_public_hash` and d.`valid_from` = c.`valid_from` 
when matched then
update set
c.`customer_sk`=d.`customer_sk`
'''

mergeDf = spark.sql(mergeQuery)
